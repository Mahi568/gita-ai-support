const API_URL = "https://script.google.com/macros/s/AKfycbxf6eFD8m88iFOPuUAxr96-aMIJtAMt6BWFobiFL7nmBgMEe94Wg_wQ9tys8wmpxG-z/exec"; async function send() { const input = document.getElementById("message"); const chat = document.getElementById("chat"); const q = input.value.trim(); if (!q) return; const userMsg = document.createElement("div"); userMsg.className = "msg user"; userMsg.textContent = q; chat.appendChild(userMsg); input.value = ""; const loading = document.createElement("div"); loading.className = "msg bot"; loading.textContent = "जवाब तैयार हो रहा है..."; chat.appendChild(loading); chat.scrollTop = chat.scrollHeight; try { const response = await fetch(API_URL, { method: "POST", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify({ message: q }) }); const data = await response.json(); loading.textContent = data.answer || "अभी जवाब नहीं मिला।"; } catch (error) { loading.textContent = "कनेक्शन में समस्या है। कृपया दोबारा कोशिश करें।"; } chat.scrollTop = chat.scrollHeight; } document.getElementById("message").addEventListener("keydown", function(e) { if (e.key === "Enter") send(); });
